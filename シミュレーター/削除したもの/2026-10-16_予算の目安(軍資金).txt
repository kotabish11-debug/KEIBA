  // ---------------- 予算の目安 ----------------
  // 各馬券のケリー比率((的中率×オッズ − 1) ÷ (オッズ − 1))の合計
  function kellySum(bets) {
    return bets.reduce((a, b) => a + (b.odds > 1 ? Math.max(0, (b.p * b.odds - 1) / (b.odds - 1)) : 0), 0);
  }
  // ケリー比率の合計の1/4を軍資金に掛ける。上限は軍資金の5%、100円単位
  function planAmount(p) {
    if (!p || !p.n) return 0;
    const share = Math.min(0.05, p.k * 0.25);
    return Math.max(100, Math.floor(bankroll * share / 100) * 100);
  }
  // 目安の金額で買い、予想どおりに決まったときの払戻
  function planPay(p, amt) { return amt && p.good ? predPay(allocate(p.good, amt), p.top) : 0; }
  // ほかのレースを、そのレースの保存値・自動更新データで一時的に計算する(表示中のレースの状態は元に戻す)
  function planOf(r) {
    if (planCache[r.id]) return planCache[r.id];
    const saved = { race, roster, horses, settings, autoData };
    try {
      race = r;
      roster = clone(r.horses);
      settings = defaultSettings();
      autoData = docs[r.id] || null;
      load(); applyAuto();
      Object.assign(settings, { style: saved.settings.style, evMin: saved.settings.evMin, runs: "5000" });
      horses = roster.filter(h => !h.scratched);
      if (horses.length < 2) return (planCache[r.id] = { k: 0, n: 0, fav: "" });
      const s = simulate();
      const { good } = pickBets(buildCandidates(s, marketProbs()));
      const top = [...Array(s.n).keys()].sort((a, b) => s.win[b] - s.win[a])[0];
      const order = [...Array(s.n).keys()].sort((a, b) => s.win[b] - s.win[a]);
      return (planCache[r.id] = { k: kellySum(good), n: good.length, fav: horses[top].name, kari: horses.some(h => h.oddsKari), good, top: order });
    } catch (e) {
      return { k: 0, n: 0, fav: "" };
    } finally {
      race = saved.race; roster = saved.roster; horses = saved.horses; settings = saved.settings; autoData = saved.autoData;
    }
  }
  function renderPlan() {
    if (!curPlan) return;
    if (document.activeElement !== $("bankroll")) $("bankroll").value = bankroll;
    const amt = planAmount(curPlan);
    $("planNow").innerHTML = amt
      ? `<div><span class="k">このレース(${esc(race.name)})の予算の目安</span><span class="v">${yen(amt)}<small>軍資金の${pct(amt / bankroll)}・${curPlan.n}点</small></span></div>
         <div><span class="k">予想どおりに決まったら戻る金額</span><span class="v">${yen10(planPay(curPlan, amt))}</span></div>
         <button type="button" class="btn" id="usePlan">この金額で買い目を出す</button>`
      : `<div><span class="k">このレース(${esc(race.name)})の予算の目安</span><span class="skip">0円(見送り)</span></div>
         <span class="small muted">期待値${esc(settings.evMin)}以上の馬券がありません</span>`;
    const { up } = sortedRaces();
    const list = up.some(r => r.id === race.id) ? up : [race].concat(up);
    let total = 0, totalPay = 0;
    $("planTable").tBodies[0].innerHTML = list.map(r => {
      const p = r.id === race.id ? curPlan : planOf(r);
      const a = planAmount(p);
      const pay = Math.floor(planPay(p, a) / 10) * 10;
      total += a; totalPay += pay;
      return `<tr class="${r.id === race.id ? "cur" : ""}">
        <td class="num">${esc(dateShort(r.iso))}</td>
        <td><button type="button" class="rbtn" data-race="${esc(r.id)}">${esc(r.place + r.no + "R " + r.name)}</button> ${gradeBadge(r.grade, true)}</td>
        <td class="r num">${a ? yen(a) : '<span class="muted">見送り</span>'}${a && p.kari ? '<span class="muted small">仮</span>' : ""}</td>
        <td class="r num">${a ? (pay ? yen(pay) : '<span class="muted small">当たらない</span>') : "－"}</td>
        <td class="r num">${a ? pct(a / bankroll) : "－"}</td>
        <td class="r num">${p.n ? p.n + "点" : "－"}</td>
        <td>${esc(p.fav || "")}</td>
      </tr>`;
    }).join("");
    $("planTable").tFoot.innerHTML = `<tr><td colspan="2">合計(${list.length}レース)</td><td class="r num">${yen(total)}</td><td class="r num">${yen(totalPay)}</td><td class="r num">${pct(total / bankroll)}</td><td colspan="2" class="muted small">「仮」は仮オッズでの目安。実オッズが出ると変わります</td></tr>`;
  }
